\subsection*{2 (a)}
Using the normalization $\mathbb{E}[M^*] = 1$, we find the stochastic discount factor $M_t^* = a - b'xR_t$ 
that satisfies $\mathbb{\hat{E}}[M^* \cdot xR] = 0$. In particular, we have:
\begin{align*}
    0 
    &= \mathbb{{E}}[M^* \cdot xR] \\
    &= \underbrace{\mathbb{{E}}[M^*]}_{\displaystyle = 1} \cdot \mathbb{E}[xR] + \text{Cov}(M^*, xR) \\
    &= \mathbb{E}[xR] - b'\text{Cov}(xR, xR) \\
    &\implies \mathbb{E}[xR] = b'\Sigma \\
    &\implies b = \Sigma^{-1}\mathbb{E}[xR] 
\end{align*}

$xR$ is a Tx10 vector of excess returns with T = 1,202 months in the sample from July 1926 to August 2026. Verifying that $\Sigma$ is invertible,
we find that:
\begin{align*}
    1 
    &= \mathbb{{E}}[M^*] \\
    &= \mathbb{E}[a - b'xR] \\
    &= a - b'\mathbb{E}[xR] \\
    &\implies a = 1 + b'\mathbb{E}[xR] \\
\end{align*}
Therefore, we are able to compute the stochastic discount factor $M_t^* = a - b'xR_t$ for each month in the sample and obtain the following scatterplot:

\section*{Question 3}
\subsection*{3 (a)}
Consider a representative investor with wealth $W_t$ in a two-period economy. Let
$R_{f,t}$ be a baseline asset. We solve the following optimization problem:
\begin{align*}
    \varpi_t^*
    &= \arg\max_{\varpi_t} \mathbb{E}_t\!\left[U(W_{t+1})\right], \\
    \text{s.t.}\quad
    W_{t+1}
    &= W_t \cdot R_{w,t+1}
     = W_t \cdot (R_{f,t+1}+\varpi_t' xR_{t+1}),
\end{align*}
where $xR_{t+1}$ is the vector of asset excess returns relative to $R_{f,t}$.

Using the power utility function
\[
    U(W)=\frac{W^{1-\gamma}}{1-\gamma},
\]
where $\gamma$ is CRRA, we rewrite the problem as
\[
    \varpi_t^*
    = \arg\max_{\varpi_t}
      \mathbb{E}_t\!\left[
        \frac{
          \left(W_t \cdot \left(R_{f,t+1}+\varpi_t' xR_{t+1}\right)\right)^{1-\gamma}
        }{1-\gamma}
      \right].
\]

Thus, the first-order condition of the maximization problem is
\begin{align*}
    \mathbb{E}_t\!\left[
      \frac{
        (1-\gamma) [W_t \cdot
        \left(R_{f,t+1}+\varpi_t' xR_{t+1}\right)]^{-\gamma}
        \cdot xR_{t+1}
      }{1-\gamma}
    \right] &= 0 \\
    \implies
    \mathbb{E}_t\!\left[
      \left(R_{f,t+1}+\varpi_t' xR_{t+1}\right)^{-\gamma}
      \cdot xR_{t+1}
    \right] &= 0 \\
    \implies
    \mathbb{E}_t\!\left[
      R_{m,t+1}^{-\gamma} \cdot xR_{t+1}
    \right] &= 0.
\end{align*}
As a result, the SDF $M_t=R_{m,t+1}^{-\gamma}$ prices the vector of excess returns.

forecast of $xR_{e,t+1}$, $\hat{\mathbb{E}}_{t}^{OS}[xR_e]$, and construct a managed portfolio 
using $z_t = \hat{\mathbb{E}}_{t}^{OS}[xR_e]/ \sigma_{e}^{2}$ as a trading signal based on the dividend-price ratio. 
Thus, given 1 parameter $\gamma$, we have 2 moment conditions:
\begin{align*}
    \mathbb{E}\!\left[
      R_{w,t}^{-\gamma} \cdot xR_{w,t}
    \right] &= 0, \\
    \mathbb{E}\!\left[
      R_{w,t}^{-\gamma} \cdot (z_{t-1} \cdot xR_{w,t})
    \right] &= 0.
\end{align*}

Using an identity weighting matrix, we have the following optimization problem:
\[
    \hat{\gamma} = \arg\min_{\gamma} g_{T}(\gamma)'g_{T}(\gamma),
\]

R_{w,t}^{-\gamma} \cdot xR_{w,t} \\
R_{w,t}^{-\gamma} \cdot (z_{t-1} \cdot xR_{w,t})
\end{bmatrix}$.

\section*{Question 4}
\subsection*{4 (a)}
The SDF = $M_t = R_{w,t+1}^{-\gamma}$ is non-linear in $\gamma$ and can be written as $f(R_{w,t+1}, \gamma)$. 
Using first-order Taylor approximation, which gives $M_{t+1} = f(F_{t+1}, \theta) \approx f(\mathbb{E}_{t}[F], \theta) + \partial_{F} f(\mathbb{E}_{t}[F], \theta)' (F_{t+1} - \mathbb{E}_{t}[F])$,
we can linearize the SDF as follows:

\begin{align*}
    M_{t+1}
    &= f(R_{w,t+1},\gamma) \\
    &\approx f\!\left(\mathbb{E}_t[R_{w,t+1}],\gamma\right)
    + \frac{\partial f}{\partial R}
      \!\left(\mathbb{E}_t[R_{w,t+1}],\gamma\right)
      \left(R_{w,t+1}-\mathbb{E}_t[R_{w,t+1}]\right) 
      \equiv M_{t+1}^{\mathrm{Linear}} \\
    &= \left(\mathbb{E}_t[R_{w,t+1}]\right)^{-\gamma}
    - \gamma\left(\mathbb{E}_t[R_{w,t+1}]\right)^{-\gamma-1}
      \left(R_{w,t+1}-\mathbb{E}_t[R_{w,t+1}]\right) \\
    &= \left(\mathbb{E}_t[R_{w,t+1}]\right)^{-\gamma}
       \left(
       1-\gamma
       \frac{R_{w,t+1}-\mathbb{E}_t[R_{w,t+1}]}
            {\mathbb{E}_t[R_{w,t+1}]}
       \right) \\
    &= (1+\gamma)\left(\mathbb{E}_t[R_{w,t+1}]\right)^{-\gamma}
    - \gamma\left(\mathbb{E}_t[R_{w,t+1}]\right)^{-\gamma-1}
      R_{w,t+1} \\
    &= \mathbb{E}_t\!\left[M_{t+1}^{\mathrm{Linear}}\right]
\left(
(1+\gamma)
-\frac{\gamma}{\mathbb{E}_t[R_{w,t+1}]}R_{w,t+1}
\right) \\ 
    &= \mathbb{E}_t\!\left[M_{t+1}^{\mathrm{Linear}}\right]
\left(a_t-b_tR_{w,t+1}\right)
\end{align*}
where
\[
a_t=1+\gamma,
\qquad
b_t
= 
-\frac{
\partial_R f\!\left(\mathbb{E}_t[R_{w,t+1}],\gamma\right)
}{
f\!\left(\mathbb{E}_t[R_{w,t+1}],\gamma\right)
}
=\frac{\gamma}{\mathbb{E}_t[R_{w,t+1}]},
\qquad
\mathbb{E}_t\!\left[M_{t+1}^{\mathrm{Linear}}\right]
=
\left(\mathbb{E}_t[R_{w,t+1}]\right)^{-\gamma}.
\]

A disadvantage of the linearized SDF is that $a_t$ and $b_t$ are time-varying. In particular, 
$b_t$ is a function of the conditional expectation of the market return $\mathbb{E}_t[R_{w,t+1}]$,
which is not observable. Therefore, errors in estimating $\mathbb{E}_t[R_{w,t+1}]$ can distort $b_t$,
thus affecting the pricing power of the linearized SDF.
